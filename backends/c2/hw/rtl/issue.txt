

A software side of the compute core task issuer should look like this:


  float dxkernel = dx * (float)#kernel_width(mand_core);
  #issue_threads_sync(mand_core, 4, (dx){cxstep = dx, ix = 0}, vram_blit) {
     for (y = 0; y < 480; y++) {
       for (x = 0; x < 640; x+= #kernel_width(mand_core)) {
          #emit_task( cx0 = cx, cy = cy, blit_destination = (y<<16+x) );
          cx += dxkernel;
       }
       cy += dy;
     }
  }


This will generate an issue core with 4 mand_core instances,
wrapped into a newly generated mand_core_wrapper module with fixed args
cxstep and ix, and variable args cx0 and cy; also adding blit_destination
packing x and y into a 32-bit vector.



Macros expanded will look like this:

float dxkernel = dx * (float)#kernel_width(mand_core);
#compute_wrapper_instance(cores=4, packet=32+64, args=64, addr=16, outwidth=4, stages=#kernel_width(mand_core),
                          issue=issue_mand_core, constarg=constarg, vararg=vararg, blitdest=blitdest, issue_available=issue_available,
                          issue_req=issue_req,issue_iddle=issue_iddle);
{
 #setup_constargs(issue_mand_core, constarg, dx, (int32)0);

 for (y = 0; y < 480; y++) {
       for (x = 0; x < 640; x+= #kernel_width(mand_core)) {
          #emit_task_inner( vararg = vararg, issue_req=issue_req,
                            cx0 = cx, cy = cy, blit_destination = (y<<16+x) );
          cx += dxkernel;
       }
       cy += dy;
 }
 #wait_task_sync(issue_mand_core);

}
